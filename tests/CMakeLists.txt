cmake_minimum_required(VERSION 3.22)
project(framework_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(pid_tests pid_tests.cpp ../libs/src/pid.cpp ../libs/src/math.cpp)
target_include_directories(pid_tests PRIVATE ../libs/include)
target_compile_options(pid_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME pid COMMAND pid_tests)
add_executable(imu_tests imu_tests.cpp ../modules/src/BMI088.cpp ../modules/src/LED.cpp)
target_include_directories(imu_tests PRIVATE ../bsp/include ../modules/include)
target_compile_options(imu_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME imu COMMAND imu_tests)
add_executable(motor_tests motor_tests.cpp ../modules/src/DJIMotorHandler.cpp ../modules/src/M2006.cpp ../libs/src/pid.cpp ../libs/src/math.cpp)
target_include_directories(motor_tests PRIVATE ../bsp/include ../modules/include ../libs/include)
target_compile_options(motor_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME motor COMMAND motor_tests)
add_executable(service_tests service_tests.cpp
    ../threads/src/threads.cpp ../threads/src/controlthread.cpp ../threads/src/alivethread.cpp
    ../modules/src/DJIMotorHandler.cpp ../modules/src/M2006.cpp ../modules/src/LED.cpp
    ../libs/src/pid.cpp ../libs/src/math.cpp)
target_include_directories(service_tests PRIVATE service_mocks ../threads/include ../bsp/include ../modules/include ../libs/include)
target_compile_definitions(service_tests PRIVATE APP_ENABLE_IMU=1 APP_ENABLE_CONTROL=1)
target_compile_options(service_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME service COMMAND service_tests)
add_executable(bsp_tests bsp_tests.cpp ../bsp/src/bsp_usart.cpp ../bsp/src/bsp_can.cpp)
target_include_directories(bsp_tests PRIVATE mocks ../bsp/include)
target_compile_options(bsp_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME bsp COMMAND bsp_tests)
